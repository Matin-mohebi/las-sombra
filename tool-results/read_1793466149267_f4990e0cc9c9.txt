    77→  /** merge geometries with normalized attributes (index + color) */
    78→  private mergeAll(geos: THREE.BufferGeometry[], defaultColor = '#ffffff'): THREE.BufferGeometry {
    79→    const out = geos.map(g => {
    80→      const gg = g.index ? g.toNonIndexed() : g
    81→      if (!gg.attributes.color) {
    82→        const n = gg.attributes.position.count
    83→        const c = new THREE.Color(defaultColor)
    84→        const col = new Float32Array(n * 3)
    85→        for (let i = 0; i < n; i++) { col[i * 3] = c.r; col[i * 3 + 1] = c.g; col[i * 3 + 2] = c.b }
    86→        gg.setAttribute('color', new THREE.BufferAttribute(col, 3))
    87→      }
    88→      return gg
    89→    })
    90→    return mergeGeometries(out, false)!
    91→  }

   116→  private roadGeo(x: number, z: number, len: number, vert: boolean, w = ROAD_W) {
   117→    const g = new THREE.PlaneGeometry(w, len)
   118→    const uv = g.attributes.uv as THREE.BufferAttribute
   119→    for (let i = 0; i < uv.count; i++) uv.setY(i, uv.getY(i) * (len / 16))
   120→    g.rotateX(-Math.PI / 2)
   121→    if (!vert) g.rotateY(Math.PI / 2)
   122→    g.translate(x, 0.06, z)
   123→    return g
   124→  }

   272→  private addWindows(x: number, z: number, w: number, d: number, h: number) {
   273→    const floors = Math.floor((h - 4) / 3.1)
   274→    if (floors < 1) return
   275→    const faces = [
   276→      { ry: 0, len: w }, { ry: Math.PI, len: w }, { ry: Math.PI / 2, len: d }, { ry: -Math.PI / 2, len: d },
   277→    ]
   278→    for (const f of faces) {
   279→      const cols = Math.floor((f.len - 3) / 2.9)
   280→      for (let fl = 0; fl < floors; fl++) {
   281→        for (let c = 0; c < cols; c++) {
   282→          if (Math.random() < 0.18) continue
   283→          const y = 2.4 + fl * 3.1
   284→          const off = -f.len / 2 + 2.2 + c * 2.9
   285→          const m = new THREE.Matrix4()
   286→          const pos = f.ry === 0 ? new THREE.Vector3(x + off, y, z + d / 2 + 0.07)
   287→            : f.ry === Math.PI ? new THREE.Vector3(x + off, y, z - d / 2 - 0.07)
   288→            : f.ry === Math.PI / 2 ? new THREE.Vector3(x + w / 2 + 0.07, y, z + off)
   289→            : new THREE.Vector3(x - w / 2 - 0.07, y, z + off)
   290→          m.compose(pos, new THREE.Quaternion().setFromEuler(new THREE.Euler(0, f.ry, 0)), new THREE.Vector3(1, 1, 1))
   291→          this.windowsData.push({ m })
   292→        }
   293→      }
   294→    }
   295→  }

   312→  private buildPark(cx: number, cz: number, uw: number, ud: number) {
   313→    // plaza with trees + fountain

   315→    slab.rotation.x = -Math.PI / 2

   318→    const geos: THREE.BufferGeometry[] = []

   323→    // trees
   324→    const treeTrunk = new THREE.CylinderGeometry(0.22, 0.3, 3, 6)
   325→    const treeTop = new THREE.IcosahedronGeometry(1.9, 0)
   326→    for (let i = 0; i < 8; i++) {

   330→      const tz = cz + Math.sin(a) * r
   331→      geos.push(treeTrunk.clone().translate(tx, 1.5, tz))
   332→      const top = treeTop.clone().translate(tx, 4, tz)
   333→      this.tintGeo(top, new THREE.Color('#4d6b35').offsetHSL(0, 0, rand(-0.05, 0.06)))
   334→      geos.push(top)
   335→      this.colliders.push({ x: tx, z: tz, hw: 0.5, hd: 0.5, h: 3 })
   336→    }
   337→    const merged = this.mergeAll(geos, '#5a4632')
   338→    const m = new THREE.MeshLambertMaterial({ vertexColors: true })
   339→    this.group.add(new THREE.Mesh(merged, m))

   350→  private buildLamps() {
   351→    const lampPts: { x: number; z: number; ry: number }[] = []
   352→    for (const rx of V_ROADS) {
   353→      for (let z = -860; z <= -340; z += 105) {
   354→        if (H_ROADS.some(h => Math.abs(h - z) < 12)) continue
   355→        const side = (Math.round(z / 105) % 2 === 0) ? 1 : -1
   356→        lampPts.push({ x: rx + side * 8.6, z, ry: side > 0 ? Math.PI : 0 })
   357→      }
   358→    }
   359→    for (const rz of H_ROADS) {
   360→      for (let x = -260; x <= 260; x += 104) {
   361→        if (V_ROADS.some(v => Math.abs(v - x) < 12)) continue
   362→        const side = (Math.round(x / 104) % 2 === 0) ? 1 : -1
   363→        lampPts.push({ x, z: rz + side * 8.6, ry: side > 0 ? -Math.PI / 2 : Math.PI / 2 })
   364→      }
   365→    }
   366→    // highway lamps
   367→    for (let z = -240; z <= 560; z += 100) {
   368→      const side = (Math.round(z / 100) % 2 === 0) ? 1 : -1
   369→      lampPts.push({ x: side * 8.6, z, ry: side > 0 ? Math.PI : 0 })
   370→    }
   371→    // town lamps
   372→    for (let x = -180; x <= 180; x += 72) {
   373→      lampPts.push({ x, z: TOWN.streetZ + (x % 144 === 0 ? 9 : -9), ry: x % 144 === 0 ? Math.PI : 0 })
   374→    }
   375→
   376→    const poleGeos: THREE.BufferGeometry[] = []
   377→    const headGeos: THREE.BufferGeometry[] = []
   378→    for (const p of lampPts) {
   379→      poleGeos.push(new THREE.CylinderGeometry(0.09, 0.13, 7, 6).translate(p.x, 3.5, p.z))
   380→      const armLen = 1.7
   381→      // arm points toward road center (opposite of side offset)
   382→      const dx = Math.sin(p.ry + Math.PI)
   383→      const dz = Math.cos(p.ry + Math.PI)
   384→      const hx = p.x + dx * armLen
   385→      const hz = p.z + dz * armLen
   386→      poleGeos.push(
   387→        new THREE.BoxGeometry(0.1, 0.1, armLen)
   388→          .rotateY(Math.atan2(dx, dz))
   389→          .translate(p.x + dx * armLen / 2, 6.9, p.z + dz * armLen / 2),
   390→      )
   391→      headGeos.push(new THREE.BoxGeometry(0.55, 0.22, 0.9).rotateY(Math.atan2(dx, dz) + Math.PI / 2).translate(hx, 6.8, hz))
   392→      this.lamps.push(new THREE.Vector3(hx, 6.6, hz))
   393→    }
   394→    const poleMat = new THREE.MeshLambertMaterial({ color: 0x2c2e33 })
   395→    this.group.add(new THREE.Mesh(this.mergeAll(poleGeos.filter(Boolean)), poleMat))
   396→    this.lampHeadMat = new THREE.MeshLambertMaterial({ color: 0x554d3a, emissive: 0xffd88a, emissiveIntensity: 0 })
   397→    this.group.add(new THREE.Mesh(this.mergeAll(headGeos), this.lampHeadMat))
   398→
   399→    // glow points
   400→    const gGeo = new THREE.BufferGeometry()

   697→  /* ================= traffic graph ================= */
   698→  private buildGraph() {
   699→    const nodes: GraphNode[] = []
   700→    const idx = (x: number, z: number) => {
   701→      const i = nodes.findIndex(n => n.x === x && n.z === z)
   702→      if (i >= 0) return i
   703→      nodes.push({ x, z })
   704→      return nodes.length - 1
   705→    }
   706→    const edges: [number, number][] = []
   707→    const link = (a: number, b: number) => { edges.push([a, b]) }
   708→    // grid
   709→    const gridIdx: number[][] = []
   710→    for (let ri = 0; ri < H_ROADS.length; ri++) {
   711→      gridIdx[ri] = []
   712→      for (let ci = 0; ci < V_ROADS.length; ci++) {
   713→        gridIdx[ri][ci] = idx(V_ROADS[ci], H_ROADS[ri])
   714→      }
   715→    }
   716→    for (let ri = 0; ri < H_ROADS.length; ri++) {
   717→      for (let ci = 0; ci < V_ROADS.length; ci++) {
   718→        if (ci < V_ROADS.length - 1) link(gridIdx[ri][ci], gridIdx[ri][ci + 1])
   719→        if (ri < H_ROADS.length - 1) link(gridIdx[ri][ci], gridIdx[ri + 1][ci])
   720→      }
   721→    }
   722→    // highway chain south of city
   723→    const hwyZ = [-300, -160, 0, 160, 320, 480, 600]
   724→    let prev = idx(0, -300)
   725→    for (let i = 1; i < hwyZ.length; i++) {
   726→      const n = idx(0, hwyZ[i])
   727→      link(prev, n)
   728→      prev = n
   729→    }
   730→    // town street
   731→    const t1 = idx(-200, 600), t2 = idx(0, 600), t3 = idx(200, 600)
   732→    link(t1, t2); link(t2, t3)
   733→
   734→    // adjacency
   735→    const adj: number[][] = nodes.map(() => [])
   736→    for (const [a, b] of edges) { adj[a].push(b); adj[b].push(a) }
   737→    this.graph = { nodes, adj }
   738→  }
